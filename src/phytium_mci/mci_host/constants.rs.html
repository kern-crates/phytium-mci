130pub(crate) enum MCIHostCurrentState {
131    Idle = 0,
132    Ready = 1,
133    Identification = 2,
134    Standby = 3,
135    Transfer = 4,
136    Data = 5,
137    Receive = 6,
138    Programming = 7,
139    Disconnect = 8,
140}
141
142impl MCIHostCurrentState {
143    pub(crate) fn current_state(state: u32) -> Self {
144        let state = (state & 0x00001E00) >> 9;
145        match state {
146            0 => MCIHostCurrentState::Idle,
147            1 => MCIHostCurrentState::Ready,
148            2 => MCIHostCurrentState::Identification,
149            3 => MCIHostCurrentState::Standby,
150            4 => MCIHostCurrentState::Transfer,
151            5 => MCIHostCurrentState::Data,
152            6 => MCIHostCurrentState::Receive,
153            7 => MCIHostCurrentState::Programming,
154            8 => MCIHostCurrentState::Disconnect,
155            _ => MCIHostCurrentState::Idle,
156        }
157    }
158}
