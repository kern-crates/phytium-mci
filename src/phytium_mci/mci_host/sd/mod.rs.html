1//! SD Card Driver.
2//!
3//! This module provides a high-level driver for SD memory cards, implementing
4//! the SD specification (versions 1.0-3.0). It supports SDSC, SDHC, and SDXC
5//! cards with various bus widths and transfer modes.
6//!
7//! # Supported Card Types
8//!
9//! - **SDSC** (Standard Capacity SD Card): ≤ 2GB
10//! - **SDHC** (High Capacity SD Card): 2GB - 32GB
11//! - **SDXC** (Extended Capacity SD Card): 32GB - 2TB
12//!
13//! # Features
14//!
15//! - **Bus Widths**: 1-bit and 4-bit data bus
16//! - **Voltage**: 3.3V (default) and 1.8V (UHS-I)
17//! - **Speed Modes**: Default, High Speed, SDR12, SDR25, SDR50, SDR104
18//! - **Transfer Modes**: DMA and PIO
19//! - **Card Detection**: Automatic card detection
20//! - **Write Protection**: Card-level write protection support
21//!
22//! # Card Information
23//!
24//! The driver reads and provides access to:
25//! - **CID** (Card Identification): Manufacturer, product name, serial number
26//! - **CSD** (Card Specific Data): Capacity, block size, speed class
27//! - **SCR** (SD Configuration Register): SD version, bus width support
28//! - **SSR** (SD Status): Card status information
29//!
30//! # Example
31//!
32//! ```rust,ignore
33//! use phytium_mci::sd::SdCard;
34//! use phytium_mci::IoPad;
35//! use core::ptr::NonNull;
36//!
37//! // Create IOPAD instance
38//! let iopad = unsafe { IoPad::new(NonNull::new_unchecked(0x2800_0000 as *mut u8)) };
39//!
40//! // Create SD card instance
41//! let mut sdcard = unsafe {
42//!     SdCard::new(
43//!         NonNull::new_unchecked(0x2800_1000 as *mut u8),
44//!         iopad
45//!     )
46//! };
47//!
48//! // Initialize the card
49//! sdcard.init(NonNull::new_unchecked(0x2800_1000 as *mut u8))?;
50//!
51//! // Read blocks
52//! let mut buffer = Vec::new();
53//! sdcard.read_blocks(&mut buffer, 0, 1)?;
54//!
55//! // Get card information
56//! println!("Capacity: {} MB", sdcard.capacity() / (1024 * 1024));
57//! println!("Block size: {} bytes", sdcard.block_size());
58//! println!("Block count: {}", sdcard.block_count());
59//! ```

130/// SD Card driver.
131///
132/// `SdCard` provides a high-level interface for SD memory cards,
133/// handling initialization, data transfer, and card information access.
134///
135/// # Card State
136///
137/// The driver maintains the following card state:
138/// - `version` - SD specification version (1.0-3.0)
139/// - `flags` - Card capability flags
140/// - `current_timing` - Current timing mode
141/// - `operation_voltage` - Current operating voltage
142/// - `block_count` - Total number of blocks
143///
144/// # Card Information
145///
146/// Card identification and configuration data is available through:
147/// - `cid` - Card Identification Register
148/// - `csd` - Card Specific Data Register
149/// - `scr` - SD Configuration Register
150/// - `stat` - SD Status Register
151///
152/// # Initialization
153///
154/// The card must be initialized before use. The initialization process:
155/// 1. Resets the card to idle state
156/// 2. Negotiates operating voltage
157/// 3. Queries card capacity and capabilities
158/// 4. Configures bus width and timing
159/// 5. Reads card information registers

162    usr_param: SdUsrParam,
163    version: SdSpecificationVersion,
164    flags: SdCardFlag,

167    driver_strength: SdDriverStrength,
168    max_current: SdMaxCurrent,
169    operation_voltage: MCIHostOperationVoltage,
170    cid: SdCid,
171    csd: SdCsd,
172    scr: SdScr,
173    stat: SdStatus,
174}

177    /// Creates a new SD card instance
178    ///
179    /// # Arguments
180    ///
181    /// * `addr` - Base address of the MCI controller registers
182    /// * `iopad` - I/O pad instance for signal delay tuning
183    ///
184    /// # Panics
185    ///
186    /// Panics if internal buffer allocation fails or SDIF/SDMMC configuration fails.
187    pub fn new(addr: NonNull<u8>, iopad: IoPad) -> Self {
188        osa_init();
189
190        let mci_host_config = MCIHostConfig::new();
191
192        // Assemble base
193        let internal_buffer = match PoolBuffer::new(
194            mci_host_config.max_trans_size,
195            mci_host_config.def_block_size,
196        ) {
197            Err(e) => panic!("Failed to allocate internal buffer, err: {:?}", e),
198            Ok(buffer) => buffer,
199        };
200        let base = MCICardBase::from_buffer(internal_buffer);
201        info!(
202            "Internal buffer@0x{:p}, length = 0x{}",
203            base.internal_buffer.addr().as_ptr(),
204            base.internal_buffer.size()
205        );
206
207        // Assemble host
208        let desc_num = mci_host_config.max_trans_size / mci_host_config.def_block_size;
209        let sdif_device = SDIFDev::new(addr, desc_num);
210        sdif_device.iopad_set(iopad);
211        let host = MCIHost::new(Box::new(sdif_device), mci_host_config);
212        let host_type = host.config.host_type;
213
214        // Initially assemble SdCard
215        let mut sd_card = SdCard::from_base(base);
216        sd_card.base.host = Some(host);
217
218        if host_type == MCIHostType::SDIF {
219            if sd_card.sdif_config().is_err() {
220                panic!("Config fail!");
221            }
222        } else if sd_card.sdmmc_config().is_err() {
223            panic!("Config fail!");
224        }
225
226        sd_card
227    }

229    fn sdif_config(&mut self) -> MCIHostStatus {
230        let mut card_cd = MCIHostCardDetect::new();
231
232        card_cd.typ = MCIHostDetectCardType::ByHostCD;
233        card_cd.cd_debounce_ms = 10;
234
235        let card_cd = Rc::new(card_cd);
236
237        let usr_param = &mut self.usr_param;
238
239        usr_param.power_off_delay_ms = 0;
240        usr_param.power_on_delay_ms = 0;
241
242        let capability = MCIHostCapability::SUSPEND_RESUME
243            | MCIHostCapability::BIT4_DATA_WIDTH
244            | MCIHostCapability::BIT8_DATA_WIDTH
245            | MCIHostCapability::DETECT_CARD_BY_DATA3
246            | MCIHostCapability::DETECT_CARD_BY_CD
247            | MCIHostCapability::AUTO_CMD12
248            | MCIHostCapability::DRIVER_TYPE_C
249            | MCIHostCapability::SET_CURRENT;
250        let capability = capability.bits() | MCIHostCapabilityExt::BIT8_WIDTH.bits();
251
252        usr_param.capability = capability;
253
254        self.base.no_interal_align = false;
255
256        let host = self.base.host.as_mut().ok_or(MCIHostError::HostNotReady)?;
257
258        if host.config.is_uhs_card {
259            let mut io_voltage = SdIoVoltage::new();
260
261            io_voltage.typ_set(SdIoVoltageCtrlType::ByHost);
262            io_voltage.set_func(None);
263
264            usr_param.io_voltage = Some(io_voltage);
265
266            let capability = MCIHostCapability::VOLTAGE_3V3
267                | MCIHostCapability::VOLTAGE_1V8
268                | MCIHostCapability::HIGH_SPEED
269                | MCIHostCapability::SDR104
270                | MCIHostCapability::SDR50;
271
272            host.capability = capability;
273        } else {
274            usr_param.io_voltage = None;
275
276            let mut capability = MCIHostCapability::VOLTAGE_3V3;
277
278            if host.config.card_clock >= SD_CLOCK_50MHZ {
279                capability |= MCIHostCapability::HIGH_SPEED;
280            }
281
282            host.capability = capability;
283        }
284
285        usr_param.max_freq = host.config.card_clock;
286
287        self.usr_param.cd = Some(card_cd.clone());
288
289        host.max_block_count
290            .set(host.config.max_trans_size as u32 / host.config.def_block_size as u32);
291        host.max_block_size = MCI_HOST_MAX_BLOCK_LENGTH;
292        host.source_clock_hz = 1200000000;
293        host.cd = Some(card_cd.clone());
294
295        Ok(())
296    }

323    /// Initializes the SD card
324    ///
325    /// Performs complete card initialization including host initialization,
326    /// card detection, and card identification.
327    ///
328    /// # Arguments
329    ///
330    /// * `addr` - Base address of the MCI controller registers
331    ///
332    /// # Returns
333    ///
334    /// Returns `Ok(())` if initialization succeeds, or an error if:
335    /// - Card detection fails
336    /// - Card initialization fails
337    /// - Host initialization fails
338    pub fn init(&mut self, addr: NonNull<u8>) -> MCIHostStatus {
339        debug!("sd card initializing");
340
341        let status = if !self.base.is_host_ready {
342            self.host_init(addr)
343        } else {
344            /* reset host if it's ready */
345            self.host_do_reset()
346        };
347
348        if status.is_ok() {
349            /* check if card is presented */
350            if self.polling_card_insert(SDStatus::Inserted).is_err() {
351                info!("Polling card failed !!!");
352                return Err(MCIHostError::CardDetectFailed);
353            } else {
354                /* start card init process */
355                info!("Start card identification");
356                if let Err(err) = self.card_init() {
357                    warn!("SD card init failed !!! {:?}", err);
358                    return Err(MCIHostError::CardInitFailed);
359                }
360            }
361        }
362
363        info!("SD init finished, error = {:?}", status);
364        status
365    }

381    fn card_init_proc(&mut self) -> MCIHostStatus {
382        info!("card init proc");
383        /* reset variables */
384        self.flags = SdCardFlag::empty();
385        /* set DATA bus width */
386        let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
387        host.dev.card_bus_width_set(MCIHostBusWdith::Bit1);
388        /*set card freq to 400KHZ*/
389        self.base.bus_clk_hz = host.dev.card_clock_set(MCI_HOST_CLOCK_400KHZ, host);
390
391        /* probe bus voltage */
392        if self.bus_voltage_prob().is_err() {
393            return Err(MCIHostError::SwitchVoltageFail);
394        }
395
396        /* Read the card's CID (card identification register) */
397        /* Initialize card if the card is SD card. */
398        if self.all_cid_send().is_err() {
399            /* CMD2 */
400            return Err(MCIHostError::AllSendCidFailed);
401        }
402
403        /*
404         * Request new relative card address. This moves the card from
405         * identification mode to data transfer mode
406         */
407        if self.rca_send().is_err() {
408            /* ACMD3 */
409            return Err(MCIHostError::SendRelativeAddressFailed);
410        }
411
412        /* Card has entered data transfer mode. Get card specific data register */
413        if self.csd_send().is_err() {
414            /* CMD9 */
415            return Err(MCIHostError::SendCsdFailed);
416        }
417
418        /* Move the card to transfer state (with CMD7) to run remaining commands */
419        if self.card_select(true).is_err() {
420            /* CMD7 */
421            return Err(MCIHostError::SelectCardFailed);
422        }
423
424        /* Set to max frequency in non-high speed mode. */
425        /*
426         * With card in data transfer state, we can set SD clock to maximum
427         * frequency for non high speed mode (25Mhz)
428         */
429        let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
430        self.base.bus_clk_hz = host.dev.card_clock_set(SD_CLOCK_25MHZ, host);
431
432        /* Read SD SCR (SD configuration register),
433         * to get supported bus width
434         */
435        if self.scr_send().is_err() {
436            /* ACMD51 */
437            return Err(MCIHostError::SendScrFailed);
438        }
439
440        /*
441         * Init UHS capable SD card. Follows figure 3-16 in physical layer specification.
442         */
443        /* Set to 4-bit data bus mode. */
444        if self.flags.contains(SdCardFlag::Support4BitWidth) {
445            /* Raise bus width to 4 bits */
446            warn!("card support 4 bit width");
447            if self.data_bus_width_set(MCIHostBusWdith::Bit4).is_err() {
448                /* ACMD6 */
449                return Err(MCIHostError::SetDataBusWidthFailed);
450            }
451            let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
452            host.dev.card_bus_width_set(MCIHostBusWdith::Bit4);
453        }
454
455        /* try to get card current status */
456        if self.status_read().is_err() {
457            /* ACMD13 */
458            return Err(MCIHostError::SendScrFailed);
459        }
460
461        /* set block size */
462        if self.block_size_set(self.base.block_size).is_err() {
463            /* CMD16 */
464            return Err(MCIHostError::SetCardBlockSizeFailed);
465        }
466
467        /* SDR104, SDR50, and DDR50 mode need tuning */
468        if self.bus_timing_select().is_err() {
469            return Err(MCIHostError::SwitchBusTimingFailed);
470        }
471
472        self.card_dump();
473
474        Ok(())
475    }

477    fn bus_voltage_prob(&mut self) -> MCIHostStatus {
478        info!("probing bus voltage");
479        /* 3.3V voltage should be supported as default */
480        let mut acmd41_argument =
481            { MCIHostOCR::VDD_29_30 | MCIHostOCR::VDD_32_33 | MCIHostOCR::VDD_33_34 };
482
483        /*
484         * If card is high capacity (SDXC or SDHC), and supports 1.8V signaling,
485         * switch to new signal voltage using "signal voltage switch procedure"
486         * described in SD specification
487         */
488        if let Some(io_voltage) = self.usr_param.io_voltage.as_ref() {
489            match io_voltage.typ() {
490                SdIoVoltageCtrlType::NotSupport => { /* do nothing */ }
491                _ => {
492                    let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
493                    let capability = host.capability;
494                    if capability.contains(MCIHostCapability::VOLTAGE_1V8)
495                        && (capability.contains(MCIHostCapability::SDR104)
496                            || capability.contains(MCIHostCapability::SDR50)
497                            || capability.contains(MCIHostCapability::DDR_MODE))
498                    {
499                        info!("Support 1.8v (with SDR50/SDR104/DDR mode)");
500
501                        /* allow user select the work voltage, if not select, sdmmc will handle it automatically */
502                        acmd41_argument |= MCIHostOCR::SWITCH_18_REQUEST_FLAG;
503                        /* reset to 3v3 signal voltage */
504                        if self
505                            .switch_io_voltage(MCIHostOperationVoltage::Voltage330V)
506                            .is_ok()
507                        {
508                            /* Host changed the operation signal voltage successfully, then card need power reset */
509                            self.card_power_set(false)?;
510                            self.card_power_set(true)?;
511                        }
512                    }
513                }
514            }
515        }
516
517        self.operation_voltage = MCIHostOperationVoltage::Voltage330V;
518
519        /* send card active */
520        let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
521        host.dev.card_active_send();
522        loop {
523            /* card go idle */
524            if self.go_idle().is_err() {
525                /* CMD0 */
526                return Err(MCIHostError::GoIdleFailed);
527            }
528            /* Check card's supported interface condition. */
529            if self.interface_condition_send().is_ok() {
530                /* CMD8 */
531                /* SDHC or SDXC card */
532                acmd41_argument |= MCIHostOCR::CARD_CAPACITY_SUPPORT_FLAG;
533                self.flags |= SdCardFlag::SupportSdhc;
534            } else {
535                /* SDSC card */
536                if self.go_idle().is_err() {
537                    /* make up for legacy card which do not support CMD8 */
538                    return Err(MCIHostError::GoIdleFailed);
539                }
540            }
541
542            /* Set card interface condition according to SDHC capability and card's supported interface condition. */
543            if self
544                .application_opration_condition_send(acmd41_argument.bits())
545                .is_err()
546            {
547                /* ACMD41 */
548                return Err(MCIHostError::HandShakeOperationConditionFailed);
549            }
550
551            /* check if card support 1.8V */
552            if self.flags.contains(SdCardFlag::SupportVoltage180v) {
553                if let Some(io_voltage) = self.usr_param.io_voltage.as_ref()
554                    && io_voltage.typ() == SdIoVoltageCtrlType::NotSupport
555                {
556                    break;
557                }
558
559                match self.voltage_switch(MCIHostOperationVoltage::Voltage180V) {
560                    Err(MCIHostError::SwitchVoltageFail) => {
561                        break;
562                    }
563                    /* card enters UHS-I mode and input/ouput timings are changed to SDR12 by default */
564                    Err(MCIHostError::SwitchVoltage18VFail33VSuccess) => {
565                        acmd41_argument &= !MCIHostOCR::SWITCH_18_REQUEST_FLAG;
566                        self.flags &= !SdCardFlag::SupportVoltage180v;
567                        continue;
568                    }
569                    _ => {
570                        info!("Select 1.8v");
571                        self.operation_voltage = MCIHostOperationVoltage::Voltage180V;
572                        break;
573                    }
574                }
575            }
576            break;
577        }
578
579        Ok(())
580    }

582    fn switch_io_voltage(&mut self, voltage: MCIHostOperationVoltage) -> MCIHostStatus {
583        let io_voltage = self
584            .usr_param
585            .io_voltage
586            .as_ref()
587            .ok_or(MCIHostError::Fail)?;
588        let typ = io_voltage.typ();
589
590        if typ == SdIoVoltageCtrlType::NotSupport {
591            return Err(MCIHostError::NotSupportYet);
592        }
593
594        if typ == SdIoVoltageCtrlType::ByGpio {
595            /* make sure card signal line voltage is 3.3v before initalization */
596            if let Some(func) = io_voltage.func() {
597                func(voltage);
598            }
599        } else if typ == SdIoVoltageCtrlType::ByHost {
600            let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
601            let _ = host.dev.switch_to_voltage(voltage, host);
602        } else {
603            return Err(MCIHostError::NotSupportYet);
604        }
605
606        Ok(())
607    }

609    fn host_init(&mut self, addr: NonNull<u8>) -> MCIHostStatus {
610        info!("host init");
611        let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
612        if !self.base.is_host_ready
613            && let Err(err) = host.dev.init(addr, host)
614        {
615            info!("SD host driver init failed, error = {:?}", err);
616            return Err(MCIHostError::Fail);
617        }
618
619        let cd = self
620            .usr_param
621            .cd
622            .as_ref()
623            .ok_or(MCIHostError::HostNotReady)?;
624        if cd.typ == MCIHostDetectCardType::ByGpioCD || cd.typ == MCIHostDetectCardType::ByHostDATA3
625        {
626            info!("SD card init start");
627            let _ = host.dev.card_detect_init(cd);
628        }
629
630        /* set the host status flag, after the card re-plug in, don't need init host again */
631        self.base.is_host_ready = true;
632
633        info!("host init ok");
634        Ok(())
635    }

642    fn card_power_set(&self, enable: bool) -> MCIHostStatus {
643        if self.usr_param.sd_pwr.is_some() {
644            let sd_pwr = self.usr_param.sd_pwr.unwrap();
645            sd_pwr(enable);
646        } else {
647            let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
648            host.dev.card_power_set(enable);
649        }
650
651        let power_delay = if enable {
652            if self.usr_param.power_on_delay_ms == 0 {
653                SD_POWER_ON_DELAY_MS
654            } else {
655                self.usr_param.power_on_delay_ms
656            }
657        } else if self.usr_param.power_off_delay_ms == 0 {
658            SD_POWER_OFF_DELAY_MS
659        } else {
660            self.usr_param.power_off_delay_ms
661        };
662
663        sleep(Duration::from_millis(power_delay as u64));
664        Ok(())

667    fn polling_card_insert(&self, status: SDStatus) -> MCIHostStatus {
668        let cd = self
669            .usr_param
670            .cd
671            .as_ref()
672            .ok_or(MCIHostError::HostNotReady)?;
673
674        if cd.typ == MCIHostDetectCardType::ByGpioCD {
675            let card_detect = cd.card_detected.ok_or(MCIHostError::Fail)?;
676
677            loop {
678                if card_detect() && status == SDStatus::Inserted {
679                    let cd_debounce_ms = cd.cd_debounce_ms;
680                    sleep(Duration::from_millis(cd_debounce_ms as u64));
681                    if card_detect() {
682                        break;
683                    }
684                }
685
686                if !card_detect() && status == SDStatus::Removed {
687                    break;
688                }
689            }
690        } else {
691            /* mostly advanced host not detect card by gpio, therefore follow this branch */
692            if !self.base.is_host_ready {
693                info!("SD host not ready !!!");
694                return Err(MCIHostError::Fail);
695            }
696
697            /* polling wait until card presented or timeout */
698            let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
699            if host
700                .dev
701                .card_detect_status_polling(status, u32::MAX, host)
702                .is_err()
703            {
704                info!("Polling SD card status failed !!!");
705                return Err(MCIHostError::Fail);
706            }
707        }
708        Ok(())
709    }

711    fn polling_card_status_busy(&mut self, timeout_ms: u32) -> MCIHostStatus {
712        let mut status_timeout_us = timeout_ms * 1000;
713
714        while status_timeout_us > 0 {
715            let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
716            if !host.dev.card_is_busy() {
717                if Err(MCIHostError::CardStatusIdle) == self.card_status_send() {
718                    return Err(MCIHostError::CardStatusIdle);
719                }
720            } else {
721                /* Delay 125us to throttle the polling rate */
722                sleep(Duration::from_micros(125));
723                status_timeout_us -= 125;
724            }
725        }
726        Err(MCIHostError::CardStatusBusy)
727    }

731            != self.polling_card_status_busy(SD_CARD_ACCESS_WAIT_IDLE_TIMEOUT)

778    fn bus_timing_select(&mut self) -> MCIHostStatus {
779        if self.operation_voltage != MCIHostOperationVoltage::Voltage180V {
780            /* group 1, function 1 ->high speed mode*/
781            debug!("group1");
782            match self.func_select(SdGroupNum::TimingMode, SdTimingFuncNum::SDR25HighSpeed) {
783                Ok(_) => {
784                    /* If the result isn't "switching to high speed mode(50MHZ) successfully or card doesn't support high speed
785                     * mode". Return failed status. */
786                    let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
787
788                    self.current_timing = SdTimingMode::SDR25HighSpeedMode;
789                    self.base.bus_clk_hz = host
790                        .dev
791                        .card_clock_set(max(self.usr_param.max_freq, SD_CLOCK_50MHZ), host);
792                }
793                Err(err) => {
794                    if err == MCIHostError::NotSupportYet {
795                        /* if not support high speed, keep the card work at default mode */
796                        info!("\r\nNote: High speed mode is not supported by card\r\n");
797                        return Ok(());
798                    }
799                    return Err(err);
800                }
801            }
802        } else {
803            /* card is in UHS_I mode */
804            #[allow(clippy::never_loop)]
805            loop {
806                if self.current_timing == SdTimingMode::SDR12DefaultMode {
807                    /* if timing not specified, probe card capability from SDR104 mode */
808                    self.current_timing = SdTimingMode::SDR104Mode;
809                }
810
811                if self.current_timing == SdTimingMode::SDR104Mode {
812                    let host_capability = {
813                        let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
814                        host.capability
815                    };
816                    if host_capability.contains(MCIHostCapability::SDR104) {
817                        match self.func_select(SdGroupNum::TimingMode, SdTimingFuncNum::SDR104) {
818                            Ok(_) => {
819                                let host =
820                                    self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
821                                self.current_timing = SdTimingMode::SDR104Mode;
822                                self.base.bus_clk_hz =
823                                    host.dev.card_clock_set(SD_CLOCK_208MHZ, host);
824                                break;
825                            }
826                            _ => {
827                                info!("\r\nNote: SDR104 mode is not supported\r\n");
828                                self.current_timing = SdTimingMode::SDR50Mode;
829                            }
830                        }
831                    }
832                }
833
834                if self.current_timing == SdTimingMode::SDR50Mode {
835                    let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
836                    if host.capability.contains(MCIHostCapability::SDR50) {
837                        match self.func_select(SdGroupNum::TimingMode, SdTimingFuncNum::SDR50) {
838                            Ok(_) => {
839                                let host =
840                                    self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
841                                self.current_timing = SdTimingMode::SDR50Mode;
842                                self.base.bus_clk_hz =
843                                    host.dev.card_clock_set(SD_CLOCK_100MHZ, host);
844                                break;
845                            }
846                            _ => {
847                                info!("\r\nNote: SDR50 mode is not supported\r\n");
848                                self.current_timing = SdTimingMode::SDR25HighSpeedMode;
849                            }
850                        }
851                    }
852                }
853
854                if self.current_timing == SdTimingMode::SDR25HighSpeedMode {
855                    match self.func_select(SdGroupNum::TimingMode, SdTimingFuncNum::SDR25HighSpeed)
856                    {
857                        Ok(_) => {
858                            let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
859                            self.current_timing = SdTimingMode::SDR25HighSpeedMode;
860                            self.base.bus_clk_hz = host.dev.card_clock_set(SD_CLOCK_50MHZ, host);
861                            break;
862                        }
863                        _ => {
864                            info!("\r\nNote: SDR25 high speed mode is not supported\r\n");
865                            self.current_timing = SdTimingMode::SDR12DefaultMode;
866                        }
867                    }
868                }
869
870                info!("\r\nWarning: unknown timing mode\r\n");
871                break;
872            }
873        }
874
875        /* Update io strength according to different bus frequency */
876        if self.usr_param.io_strength.is_some() {
877            let io_strength = self.usr_param.io_strength.unwrap();
878            io_strength(self.current_timing);
879        }
880
881        /* SDR50 and SDR104 mode need tuning */
882        if self.current_timing == SdTimingMode::SDR50Mode
883            || self.current_timing == SdTimingMode::SDR104Mode
884        {
885            /* execute tuning */
886            if self.execute_tuning().is_err() {
887                info!(
888                    "\r\nError: tuning failed for mode {}\r\n",
889                    self.current_timing as u32
890                );
891                return Err(MCIHostError::TuningFail);
892            }
893        }
894
895        Ok(())
896    }

898    fn func_select(&mut self, group: SdGroupNum, func: SdTimingFuncNum) -> MCIHostStatus {
899        /* check if card support CMD6 */
900        if (self.version as u32 <= SdSpecificationVersion::Version1_0 as u32)
901            || (self.csd.card_command_classes & SdCardCmdClass::Switch.bits() == 0)
902        {
903            info!("\r\nError: current card not support CMD6\r\n");
904            return Err(MCIHostError::CardNotSupport);
905        }
906
907        /* Check if card support high speed mode. */
908        let mut func_status = match self.func_swtich(SdSwitchMode::Check, group, func) {
909            Some(status) => status,
910            None => return Err(MCIHostError::TransferFailed),
911        };
912
913        /* convert to little endian sequence */
914        let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
915        let _ = host.dev.convert_data_to_little_endian(
916            &mut func_status,
917            5,
918            MCIHostDataPacketFormat::MSBFirst,
919            host,
920        );
921
922        /*
923            -functionStatus[0U]---bit511~bit480;
924            -functionStatus[1U]---bit479~bit448;
925            -functionStatus[2U]---bit447~bit416;
926            -functionStatus[3U]---bit415~bit384;
927            -functionStatus[4U]---bit383~bit352;
928            According to the "switch function status[bits 511~0]" return by switch command in mode "check function":
929            -Check if function 1(high speed) in function group 1 is supported by checking if bit 401 is set;
930            -check if function 1 is ready and can be switched by checking if bits 379~376 equal value 1;
931        */
932        let mut func_group_info = [0u16; 6];
933        func_group_info[5] = func_status[0] as u16;
934        func_group_info[4] = (func_status[1] >> 16) as u16;
935        func_group_info[3] = (func_status[1]) as u16;
936        func_group_info[2] = (func_status[2] >> 16) as u16;
937        func_group_info[1] = (func_status[2]) as u16;
938        func_group_info[0] = (func_status[3] >> 16) as u16;
939        info!("func_group_info: {:?}", func_group_info);
940
941        let current_func_status = ((func_status[3] & 0xff) << 8) | (func_status[4] >> 24);
942
943        /* check if function is support */
944        if (func_group_info[group as usize] & (1 << (func as u16)) == 0)
945            || (((current_func_status >> ((group as u32) * 4)) & 0xf) != (func as u32))
946        {
947            info!(
948                "\r\nError: function {} in group {} not support\r\n",
949                func as u32, group as u32
950            );
951            return Err(MCIHostError::CardNotSupport);
952        }
953
954        let func_status = match self.func_swtich(SdSwitchMode::Set, group, func) {
955            Some(status) => status,
956            None => return Err(MCIHostError::TransferFailed),
957        };
958
959        /* convert to little endian sequence */
960        let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
961        let mut func_status_need_convert = func_status[3..].to_vec();
962        let _ = host.dev.convert_data_to_little_endian(
963            &mut func_status_need_convert,
964            2,
965            MCIHostDataPacketFormat::MSBFirst,
966            host,
967        );
968        let mut func_status = func_status[0..3].to_vec();
969        func_status.extend_from_slice(&func_status_need_convert);
970
971        /* According to the "switch function status[bits 511~0]" return by switch command in mode "set function":
972            -check if group 1 is successfully changed to function 1 by checking if bits 379~376 equal value 1;
973        */
974        let current_func_status = ((func_status[3] & 0xff) << 8) | (func_status[4] >> 24);
975
976        if ((current_func_status >> ((group as u32) * 4)) & 0xf) != (func as u32) {
977            info!("\r\nError: switch to function {} failed\r\n", func as u32);
978            return Err(MCIHostError::SwitchFailed);
979        }
980
981        Ok(())
982    }

992    pub fn read_blocks(
993        &mut self,
994        buffer: &mut Vec<u32>,
995        start_block: u32,
996        block_count: u32,
997    ) -> MCIHostStatus {
998        buffer.clear();
999        let mut block_left = block_count;
1000        let mut block_count_one_time: u32;
1001
1002        while block_left != 0 {
1003            let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
1004            if block_left > host.max_block_count.get() {
1005                block_left -= host.max_block_count.get();
1006                block_count_one_time = host.max_block_count.get();
1007            } else {
1008                block_count_one_time = block_left;
1009                block_left = 0;
1010            }
1011
1012            /*
1013             * todo Allocating once_buffer here may bring significant performance overhead when data volume is large
1014             * But Rust cannot use pointers as flexibly as C, otherwise it may bring a lot of unsafe code
1015             * Here we compromise by using memory allocation + extend method to handle data
1016             */
1017            let len = block_count_one_time * MCI_HOST_DEFAULT_BLOCK_SIZE / 4;
1018            let mut once_buffer = vec![0u32; len as usize];
1019            if self
1020                .read(
1021                    &mut once_buffer,
1022                    start_block,
1023                    MCI_HOST_DEFAULT_BLOCK_SIZE,
1024                    block_count_one_time,
1025                )
1026                .is_err()
1027            {
1028                return Err(MCIHostError::TransferFailed);
1029            }
1030
1031            buffer.extend(once_buffer.iter());
1032        }
1033
1034        Ok(())
1035    }

1037    /// Writes multiple blocks to the SD card
1038    ///
1039    /// Writes the specified number of blocks starting from the given block address.
1040    /// Automatically handles splitting large transfers into multiple transactions.
1041    ///
1042    /// # Arguments
1043    ///
1044    /// * `buffer` - Data buffer containing blocks to write (in u32 words)
1045    /// * `start_block` - Starting block number (0-based)
1046    /// * `block_count` - Number of blocks to write
1047    ///
1048    /// # Returns
1049    ///
1050    /// Returns `Ok(())` if write succeeds, or an error if:
1051    /// - Host is not ready
1052    /// - Transfer fails
1053    pub fn write_blocks(
1054        &mut self,
1055        buffer: &mut [u32],
1056        start_block: u32,
1057        block_count: u32,
1058    ) -> MCIHostStatus {
1059        let mut block_left = block_count;
1060        let mut block_count_one_time: u32;
1061        let mut block_written_one_time = 0; // Number of blocks successfully written in one write operation
1062
1063        while block_left != 0 {
1064            let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
1065            if block_left > host.max_block_count.get() {
1066                block_count_one_time = host.max_block_count.get();
1067            } else {
1068                block_count_one_time = block_left;
1069            }
1070
1071            let len = MCI_HOST_DEFAULT_BLOCK_SIZE * block_count_one_time / 4;
1072            let mut once_buffer = vec![0u32; len as usize];
1073            let start_addr = (block_count - block_left) * MCI_HOST_DEFAULT_BLOCK_SIZE / 4;
1074            let end_addr = start_addr + block_count_one_time * MCI_HOST_DEFAULT_BLOCK_SIZE / 4;
1075            debug!(
1076                "write block(s) one time, relative addr(u32) from {} - {}, block count {}",
1077                start_addr, end_addr, block_count_one_time
1078            );
1079            once_buffer.copy_from_slice(&buffer[start_addr as usize..end_addr as usize]);
1080            if self
1081                .write(
1082                    &mut once_buffer,
1083                    start_block + block_count - block_left,
1084                    MCI_HOST_DEFAULT_BLOCK_SIZE,
1085                    block_count_one_time,
1086                    &mut block_written_one_time,
1087                )
1088                .is_err()
1089            {
1090                error!("write block(s) failed!");
1091                return Err(MCIHostError::TransferFailed);
1092            }
1093
1094            block_left -= block_count_one_time;
1095        }
1096
1097        Ok(())
1098    }

1100    fn erase(&mut self, start_block: u32, block_count: u32) -> MCIHostStatus {
1101        let mut erase_block_start = start_block;
1102        let mut erase_block_end = erase_block_start + block_count - 1;
1103
1104        if Err(MCIHostError::CardStatusIdle)
1105            != self.polling_card_status_busy(SD_CARD_ACCESS_WAIT_IDLE_TIMEOUT)
1106        {
1107            error!("Error: write failed, card status busy");
1108            return Err(MCIHostError::TransferFailed);
1109        }
1110
1111        if !self.flags.contains(SdCardFlag::SupportHighCapacity) {
1112            erase_block_start *= MCI_HOST_DEFAULT_BLOCK_SIZE;
1113            erase_block_end *= MCI_HOST_DEFAULT_BLOCK_SIZE;
1114        }
1115
1116        // Send ERASE_WRITE_BLOCK_START command to set the start block number to erase
1117        let mut command = MCIHostCmd::new();
1118        command.index_set(SdCmd::EraseWriteBlockStart as u32);
1119        command.argument_set(erase_block_start);
1120        command.response_type_set(MCIHostResponseType::R1);
1121        command.response_error_flags_set(MCIHostCardStatusFlag::ALL_ERROR_FLAG);
1122
1123        let mut content = MCIHostTransfer::new();
1124        content.set_cmd(Some(command));
1125        if let Err(e) = self.transfer(&mut content, 1) {
1126            error!("Error: send CMD32 failed with host error {:?}", e);
1127            return Err(MCIHostError::TransferFailed);
1128        }
1129
1130        // Send ERASE_WRITE_BLOCK_END command to set the end block number to erase
1131        let mut command = MCIHostCmd::new();
1132        command.index_set(SdCmd::EraseWriteBlockEnd as u32);
1133        command.argument_set(erase_block_end);
1134
1135        content.set_cmd(Some(command));
1136        if let Err(e) = self.transfer(&mut content, 0) {
1137            error!("Error: send CMD33 failed with host error {:?}", e);
1138            return Err(MCIHostError::TransferFailed);
1139        }
1140
1141        // Send ERASE command to start erase process
1142        let mut command = MCIHostCmd::new();
1143        command.index_set(MCIHostCommonCmd::Erase as u32);
1144        command.argument_set(0);
1145        command.response_type_set(MCIHostResponseType::R1b);
1146        command.response_error_flags_set(MCIHostCardStatusFlag::ALL_ERROR_FLAG);
1147
1148        content.set_cmd(Some(command));
1149        if let Err(e) = self.transfer(&mut content, 0) {
1150            error!("Error: send CMD38 failed with host error {:?}", e);
1151            return Err(MCIHostError::TransferFailed);
1152        }
1153
1154        Ok(())
1155    }

1157    fn transfer(&mut self, content: &mut MCIHostTransfer, retry: u32) -> MCIHostStatus {
1158        let mut retry = retry;
1159        let mut retuning_count = 3;
1160        loop {
1161            let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
1162            let status = host.dev.transfer_function(content, host);
1163            if status.is_ok() {
1164                break;
1165            }
1166
1167            /* if transfer data failed, send cmd12 to abort current transfer */
1168            if content.data().is_some() {
1169                let _ = self.transmission_stop();
1170                /* when transfer error occur, polling card status until it is ready for next data transfer, otherwise the
1171                 * retry transfer will fail again */
1172                if Err(MCIHostError::CardStatusIdle)
1173                    != self.polling_card_status_busy(SD_CARD_ACCESS_WAIT_IDLE_TIMEOUT)
1174                {
1175                    return Err(MCIHostError::TransferFailed);
1176                }
1177            }
1178
1179            if retry == 0 || status == Err(MCIHostError::ReTuningRequest) {
1180                if self.current_timing == SdTimingMode::SDR104Mode
1181                    || self.current_timing == SdTimingMode::SDR50Mode
1182                {
1183                    if retuning_count == 0 {
1184                        break;
1185                    }
1186                    retuning_count -= 1;
1187                    /* Perform retuning, CMD19 sends a tuning block to the host to determine sampling point.
1188                    UHS50 and UHS104 cards support CMD19 in 1.8V signaling. Sampling
1189                    clock tuning is required for UHS104 host and optional for UHS50 host. */
1190                    if self.execute_tuning().is_err() {
1191                        info!("\r\nError: retuning failed.\r\n");
1192                        return Err(MCIHostError::TuningFail);
1193                    } else {
1194                        info!("\r\nlog: retuning successfully.\r\n");
1195                        continue;
1196                    }
1197                }
1198            } else {
1199                break;
1200            }
1201
1202            if retry != 0 {
1203                retry -= 1;
1204            } else {
1205                break;
1206            }
1207        }
1208        Ok(())
1209    }
1210}

1413    /// CMD 11
1414    fn voltage_switch(&mut self, voltage: MCIHostOperationVoltage) -> MCIHostStatus {
1415        let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
1416
1417        let mut command = MCIHostCmd::new();
1418
1419        command.index_set(SdCmd::VoltageSwitch as u32);
1420        command.argument_set(0);
1421        command.response_type_set(MCIHostResponseType::R1);
1422
1423        let mut content = MCIHostTransfer::new();
1424        content.set_cmd(Some(command));
1425
1426        if host.dev.transfer_function(&mut content, host).is_err() {
1427            return Err(MCIHostError::TransferFailed);
1428        }
1429
1430        /*
1431         * Card should drive CMD and DAT[3:0] signals low at the next clock
1432         * cycle. Some cards will only drive these
1433         * lines low briefly, so we should check as soon as possible
1434         */
1435        if !host.dev.card_is_busy() {
1436            /* Delay 1ms to allow card to drive lines low */
1437            sleep(Duration::from_millis(1));
1438            if !host.dev.card_is_busy() {
1439                /* Card did not drive CMD and DAT lines low */
1440                info!("\r\nError: card not drive lines low\r\n");
1441                return Err(MCIHostError::CardStatusBusy);
1442            }
1443        }
1444
1445        /*
1446         * Per SD spec (section "Timing to Switch Signal Voltage"),
1447         * host must gate clock at least 5ms.
1448         */
1449        host.dev.card_clock_set(0, host);
1450
1451        /* switch io voltage */
1452        if self.switch_io_voltage(voltage) == Err(MCIHostError::NotSupportYet) {
1453            info!("Failed to switch SD host to 1.8V");
1454            return Err(MCIHostError::SwitchVoltageFail);
1455        }
1456
1457        /* Gate for 10ms, even though spec requires 5 */
1458        sleep(Duration::from_millis(10));
1459
1460        /* Re-acquire host instance */
1461        let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
1462
1463        /* Restart the clock */
1464        host.dev.card_clock_set(self.base.bus_clk_hz, host);
1465
1466        /*
1467         * If SD does not drive at least one of
1468         * DAT[3:0] high within 1ms, switch failed
1469         */
1470        sleep(Duration::from_millis(1));
1471
1472        if host.dev.card_is_busy() {
1473            info!("Card failed to switch voltages");
1474            return Err(MCIHostError::SwitchVoltageFail);
1475        }
1476
1477        info!("Card switched to 1.8V signaling");
1478        Ok(())
1479    }

1888            host,
1889        );
1890
1891        self.decode_status(rx_data);
1892
1893        Ok(())

1896    /// ACMD 41
1897    fn application_opration_condition_send(&mut self, argument: u32) -> MCIHostStatus {
1898        let mut command = MCIHostCmd::new();
1899
1900        command.index_set(SdAppCmd::SendOperationCondition as u32);
1901        command.argument_set(argument);
1902        command.response_type_set(MCIHostResponseType::R3);
1903
1904        let mut content = MCIHostTransfer::new();
1905        content.set_cmd(Some(command));
1906
1907        let mut i = MCI_HOST_MAX_CMD_RETRIES;
1908        while i > 0 {
1909            if self.application_cmd_send(0).is_err() {
1910                continue;
1911            }
1912
1913            let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
1914
1915            if let Err(err) = host.dev.transfer_function(&mut content, host) {
1916                info!(
1917                    "\r\nError: send CMD8 failed with host error {:?}, response {}\r\n",
1918                    err,
1919                    {
1920                        let command = content.cmd().unwrap();
1921                        let response = command.response();
1922                        response[0]
1923                    }
1924                );
1925                return Err(MCIHostError::TransferFailed);
1926            }
1927
1928            /* Wait until card exit busy state. */
1929            let command = content.cmd().unwrap();
1930            let response = command.response()[0];
1931            if response & MCIHostOCR::POWER_UP_BUSY_FLAG.bits() != 0 {
1932                /* high capacity check */
1933                if response & MCIHostOCR::HOST_CAPACITY_SUPPORT_FLAG.bits() != 0 {
1934                    self.flags |= SdCardFlag::SupportHighCapacity;
1935                    info!("capcity > 2GB, is high capacity card")
1936                }
1937
1938                /* 1.8V support */
1939                if response & MCIHostOCR::SWITCH_18_ACCEPT_FLAG.bits() != 0 {
1940                    self.flags |= SdCardFlag::SupportVoltage180v;
1941                    info!("supports 1.8v, card is UHS card")
1942                } else {
1943                    info!("only supports 3.3v, card is not UHS card")
1944                }
1945                self.base.ocr = response;
1946                return Ok(());
1947            }
1948
1949            i -= 1;
1950            sleep(Duration::from_millis(10));
1951        }
1952
1953        info!("\r\nError: send ACMD41 timeout\r\n");
1954        Ok(())
1955    }

1957    /// ACMD 51
1958    fn scr_send(&mut self) -> MCIHostStatus {
1959        if self
1960            .application_cmd_send(self.base.relative_address)
1961            .is_err()
1962        {
1963            return Err(MCIHostError::SendApplicationCommandFailed);
1964        }
1965
1966        let host = self.base.host.as_ref().ok_or(MCIHostError::HostNotReady)?;
1967
1968        let mut command = MCIHostCmd::new();
1969
1970        command.index_set(SdAppCmd::SendScr as u32);
1971        command.argument_set(0);
1972        command.response_type_set(MCIHostResponseType::R1);
1973
1974        let mut data = MCIHostData::new();
1975
1976        data.block_size_set(8);
1977        data.block_count_set(1);
1978        // We implemented 'Drop' for PoolBuffer, so we don't need to dealloc this rx_buf
1979        let rx_buf = match PoolBuffer::new(u32::BITS as usize, size_of::<u32>()) {
1980            Err(e) => panic!("ACMD51 allocate rx_buf failed, err {:?}", e),
1981            Ok(buf) => {
1982                debug!(
1983                    "ACMD51: alloc rx_buf at {:p}, size {}",
1984                    buf.addr().as_ptr(),
1985                    buf.size()
1986                );
1987                buf
1988            }
1989        };
1990        data.rx_data_set(Some(rx_buf.to_vec::<u32>().unwrap()));
1991
1992        let mut content = MCIHostTransfer::new();
1993        content.set_cmd(Some(command));
1994        content.set_data(Some(data));
1995
1996        if let Err(err) = host.dev.transfer_function(&mut content, host) {
1997            info!("\r\nError: send CMD51 failed with host error {:?}\r\n", err);
1998            return Err(err);
1999        }
2000
2001        let raw_src = content.data_mut().unwrap().rx_data_mut().unwrap();
2002        debug!("src from transfer {:?}", raw_src);
2003        /* according to spec. there are two types of Data packet format for SD card
2004        1. Usual data (8-bit width), are sent in LSB first
2005        2. Wide width data (SD Memory register), are shifted from the MSB bit,
2006            e.g. ACMD13 (SD Status), ACMD51 (SCR) */
2007        let _ = host.dev.convert_data_to_little_endian(
2008            raw_src,
2009            2,
2010            MCIHostDataPacketFormat::MSBFirst,
2011            host,
2012        );
2013
2014        /* decode scr */
2015        self.decode_scr(raw_src);
2016
2017        Ok(())
2018    }

2046

2048    }

2050    fn decode_csd(&mut self) {
2051        let csd = &mut self.csd;
2052        let rawcsd = match self.base.internal_buffer.to_vec_in_len::<u32>(4) {
2053            Err(e) => {
2054                error!(
2055                    "Construct Vec<u32> from internal_buffer failed! err: {:?}",
2056                    e
2057                );
2058                panic!();
2059            }
2060            Ok(rawcsd) => rawcsd,
2061        };
2062
2063        csd.csd_structure = ((rawcsd[3] & 0xC0000000) >> 30) as u8;
2064        csd.data_read_access_time1 = ((rawcsd[3] & 0xFF0000) >> 16) as u8;
2065        csd.data_read_access_time2 = ((rawcsd[3] & 0xFF00) >> 8) as u8;
2066        csd.transfer_speed = (rawcsd[3] & 0xFF) as u8;
2067        csd.card_command_classes = ((rawcsd[2] & 0xFFF00000) >> 20) as u16;
2068        csd.read_block_length = ((rawcsd[2] & 0xF0000) >> 16) as u8;
2069
2070        if rawcsd[2] & 0x8000 != 0 {
2071            csd.flags |= CsdFlags::READ_BLOCK_PARTIAL.bits();
2072        }
2073        if rawcsd[2] & 0x4000 != 0 {
2074            csd.flags |= CsdFlags::READ_BLOCK_PARTIAL.bits();
2075        }
2076        if rawcsd[2] & 0x2000 != 0 {
2077            csd.flags |= CsdFlags::READ_BLOCK_MISALIGN.bits();
2078        }
2079        if rawcsd[2] & 0x1000 != 0 {
2080            csd.flags |= CsdFlags::DSR_IMPLEMENTED.bits();
2081        }
2082        if csd.csd_structure == 0 {
2083            info!("   csd structure: 1.0");
2084            csd.device_size = ((rawcsd[2] & 0x3FF) << 2) | ((rawcsd[1] & 0xC0000000) >> 30);
2085            csd.read_current_vdd_min = ((rawcsd[1] & 0x38000000) >> 27) as u8;
2086            csd.read_current_vdd_max = ((rawcsd[1] & 0x7000000) >> 24) as u8;
2087            csd.write_current_vdd_min = ((rawcsd[1] & 0xE00000) >> 20) as u8;
2088            csd.write_current_vdd_max = ((rawcsd[1] & 0x1C0000) >> 18) as u8;
2089            csd.device_size_multiplier = ((rawcsd[1] & 0x38000) >> 15) as u8;
2090            /* Get card total block count and block size. */
2091            self.block_count = (csd.device_size + 1) << (csd.device_size_multiplier + 2);
2092            self.base.block_size = 1 << csd.read_block_length;
2093            if self.base.block_size > MCI_HOST_DEFAULT_BLOCK_SIZE {
2094                self.block_count *= self.base.block_size;
2095                self.base.block_size = MCI_HOST_DEFAULT_BLOCK_SIZE;
2096                self.block_count /= self.base.block_size;
2097            }
2098        } else if csd.csd_structure == 1 {
2099            info!("   csd structure: 2.0");
2100            self.base.block_size = MCI_HOST_DEFAULT_BLOCK_SIZE;
2101            csd.device_size = ((rawcsd[2] & 0x3F) << 16) | ((rawcsd[1] & 0xFFFF0000) >> 16);
2102            if csd.device_size >= 0xFFFF {
2103                info!("device size is {}, supports sdxc", csd.device_size);
2104                self.flags |= SdCardFlag::SupportSdxc;
2105            }
2106            self.block_count = (csd.device_size + 1) * 1024;
2107        } else {
2108            info!("unknown SD CSD structure version 0x{:x}", csd.csd_structure);
2109            /* not support csd version */
2110        }
2111
2112        if ((rawcsd[1] & 0x4000) >> 14) as u8 != 0 {
2113            csd.flags |= CsdFlags::ERASE_BLOCK_ENABLED.bits();
2114        }
2115
2116        csd.erase_sector_size = ((rawcsd[1] & 0x3F80) >> 7) as u8;
2117        csd.write_protect_group_size = (rawcsd[1] & 0x7F) as u8;
2118
2119        if (rawcsd[0] & 0x80000000) as u8 != 0 {
2120            csd.flags |= CsdFlags::WRITE_PROTECT_GROUP_ENABLED.bits();
2121        }
2122
2123        csd.write_speed_factor = ((rawcsd[0] & 0x1C000000) >> 26) as u8;
2124        csd.write_block_length = ((rawcsd[0] & 0x3C00000) >> 22) as u8;
2125
2126        if ((rawcsd[0] & 0x200000) >> 21) as u8 != 0 {
2127            csd.flags |= CsdFlags::WRITE_BLOCK_PARTIAL.bits();
2128        }
2129        if ((rawcsd[0] & 0x8000) >> 15) as u8 != 0 {
2130            csd.flags |= CsdFlags::FILE_FORMAT_GROUP.bits();
2131        }
2132        if ((rawcsd[0] & 0x4000) >> 14) as u8 != 0 {
2133            csd.flags |= CsdFlags::COPY.bits();
2134        }
2135        if ((rawcsd[0] & 0x2000) >> 13) as u8 != 0 {
2136            csd.flags |= CsdFlags::PERMANENT_WRITE_PROTECT.bits();
2137        }
2138        if ((rawcsd[0] & 0x1000) >> 12) as u8 != 0 {
2139            csd.flags |= CsdFlags::TEMPORARY_WRITE_PROTECT.bits();
2140        }
2141        csd.file_format = ((rawcsd[0] & 0xC00) >> 10) as u8;

2143        info!(
2144            "Card block count {}, block size {}",
2145            self.block_count, self.base.block_size
2146        );

2149    fn decode_scr(&mut self, rawscr: &[u32]) {
2150        let scr = &mut self.scr;
2151
2152        scr.scr_structure = ((rawscr[0] & 0xF0000000) >> 28) as u8;
2153        scr.sd_specification = ((rawscr[0] & 0xF000000) >> 24) as u8;
2154        if ((rawscr[0] & 0x800000) >> 23) as u8 != 0 {
2155            scr.flags |= ScrFlags::DATA_STATUS_AFTER_ERASE.bits();
2156        }
2157        scr.sd_security = ((rawscr[0] & 0x700000) >> 20) as u8;
2158        scr.sd_bus_widths = ((rawscr[0] & 0xF0000) >> 16) as u8;
2159        if ((rawscr[0] & 0x8000) >> 15) as u8 != 0 {
2160            scr.flags |= ScrFlags::SD_SPECIFICATION3.bits();
2161        }
2162        scr.extended_security = ((rawscr[0] & 0x7800) >> 10) as u8;
2163        scr.command_support = (rawscr[0] & 0x3) as u8;
2164        scr.reserved_for_manufacturer = rawscr[1];
2165        /* Get specification version. */
2166        if scr.sd_specification == 0 {
2167            info!("   SCR version: 1.0");
2168            self.version = SdSpecificationVersion::Version1_0;
2169        } else if scr.sd_specification == 1 {
2170            info!("   SCR version: 1.1");
2171            self.version = SdSpecificationVersion::Version1_1;
2172        } else if scr.sd_specification == 2 {
2173            info!("   SCR version: 2.0");
2174            self.version = SdSpecificationVersion::Version2_0;
2175            if scr.flags & ScrFlags::SD_SPECIFICATION3.bits() != 0 {
2176                info!("   SCR version: 3.0");
2177                self.version = SdSpecificationVersion::Version3_0;
2178            }
2179        } else {
2180            info!("   SCR version: unknown");
2181        }
2182        /* Check card supported bus width */
2183        if scr.sd_bus_widths & 0x4 != 0 {
2184            info!("   Card support 4-bit bus width");
2185            self.flags |= SdCardFlag::Support4BitWidth;
2186        }
2187        /* Check if card supports speed class command (CMD20) */
2188        if scr.command_support & 0x1 != 0 {
2189            info!("   Card support speed class control command");
2190            self.flags |= SdCardFlag::SupportSpeedClassControlCmd;
2191        }
2192        /* Check if card supports set block count command (CMD23) */
2193        if scr.command_support & 0x2 != 0 {
2194            info!("   Card support set block count command");
2195            self.flags |= SdCardFlag::SupportSetBlockCountCmd;
2196        }
2197    }

2200        self.stat.bus_width = ((status[0] & 0xC0000000) >> 30) as u8; /* 511-510 */

2207        self.stat.erase_size =
2208            (((status[2] & 0x000000FF) << 8) | ((status[3] & 0xFF000000) >> 24)) as u16; /* 423-408 */
2209        self.stat.erase_timeout = (((status[3] & 0x00FF0000) >> 16) as u8 & 0xFC) >> 2; /* 407-402 */

2212        self.stat.uhs_au_size = ((status[3] & 0x0000FF00) >> 8) as u8 & 0xF; /* 395-392 */
2213    }
