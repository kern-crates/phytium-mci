27#[allow(unused)]
28impl MCIHostConfig {
29    /// Creates a new host configuration with default values.
30    ///
31    /// The default configuration uses:
32    /// - DMA transfer mode (if `dma` feature enabled)
33    /// - Polling mode (if `poll` feature enabled)
34    /// - 50 MHz card clock
35    /// - 512 byte block size
36    /// - Little endian mode
37    pub fn new() -> Self {
38        let mut config = Self {
39            host_id: MCIId::MCI1,
40            host_type: MCIHostType::SDIF,
41            card_type: MCIHostCardType::MicroSD,
42            enable_irq: false,
43            enable_dma: true,
44            endian_mode: MCIHostEndianMode::Little,
45            max_trans_size: SD_MAX_RW_BLK * SD_BLOCK_SIZE,
46            def_block_size: SD_BLOCK_SIZE,
47            card_clock: SD_CLOCK_50MHZ,
48            is_uhs_card: false,
49        };
50
51        if cfg!(feature = "dma") {
52            config.host_id = MCIId::MCI1;
53            config.enable_dma = true;
54        } else if cfg!(feature = "pio") {
55            config.host_id = MCIId::MCI0;
56            config.enable_dma = false;
57        }
58
59        if cfg!(feature = "irq") {
60            config.enable_irq = true;
61        } else if cfg!(feature = "poll") {
62            config.enable_irq = false;
63        }
64
65        config
66    }
67}
