26impl MCI {
27    /// Gets the SDIF controller interrupt mask
28    ///
29    /// # Arguments
30    ///
31    /// * `tp` - Interrupt type (General or DMA)
32    ///
33    /// # Returns
34    ///
35    /// The current interrupt mask value
36    pub fn interrupt_mask_get(&self, tp: MCIIntrType) -> u32 {
37        let reg = self.config.reg();
38        let mut mask = 0;
39        if MCIIntrType::GeneralIntr == tp {
40            mask = reg.read_reg::<MCIIntMask>().bits();
41        } else if MCIIntrType::DmaIntr == tp {
42            mask = reg.read_reg::<MCIDMACIntEn>().bits();
43        }
44        mask
45    }
46
47    /// Enables or disables SDIF controller interrupts
48    ///
49    /// # Arguments
50    ///
51    /// * `tp` - Interrupt type (General or DMA)
52    /// * `set_mask` - Interrupt mask bits to modify
53    /// * `enable` - True to enable interrupts, false to disable
54    pub fn interrupt_mask_set(&self, tp: MCIIntrType, set_mask: u32, enable: bool) {
55        let mut mask = self.interrupt_mask_get(tp);
56        if enable {
57            mask |= set_mask;
58        } else {
59            mask &= !set_mask;
60        }
61        let reg = self.config.reg();
62        if MCIIntrType::GeneralIntr == tp {
63            reg.write_reg(MCIIntMask::from_bits_truncate(mask));
64        } else if MCIIntrType::DmaIntr == tp {
65            reg.write_reg(MCIDMACIntEn::from_bits_truncate(mask));
66        }
67    }
68}

70/// Interrupt handler for SDIF instance
71///
72/// Processes all pending interrupts from the MCI controller, including
73/// command completion, data transfer completion, and error conditions.
74pub fn fsdif_interrupt_handler() {
75    let reg = MCIReg::new(reg_base());
76
77    let events = reg.read_reg::<MCIRawInts>();
78    let dmac_events = reg.read_reg::<MCIDMACStatus>();
79
80    let event_mask = reg.read_reg::<MCIIntMask>();
81    let _dmac_evt_mask = reg.read_reg::<MCIDMACIntEn>();
82
83    reg.write_reg::<MCIRawInts>(events);
84    reg.write_reg::<MCIDMACStatus>(dmac_events);
85
86    // no interrupt status
87    if (events.bits() & MCIRawInts::ALL_BITS.bits() == 0)
88        && (dmac_events.bits() & MCIDMACStatus::ALL_BITS.bits() == 0)
89    {
90        return;
91    }
92
93    reg.write_reg::<IrqTempRegister>(IrqTempRegister::from_bits_truncate(0));
94
95    // no need to handle interrput
96    if (events.bits() == 0) && (dmac_events.bits() & 0x1FFF == 0) {
97        return;
98    }
99
100    // handle sdio irq
101    if (events.bits() & event_mask.bits()) & MCIRawInts::SDIO_BIT.bits() != 0 {
102        handle_sdio_interrupt();
103        return;
104    }
105
106    // handle card detect event
107    // todo Card detection related events not yet implemented
108    // if events.bits() & event_mask.bits() & MCIRawInts::CD_BIT.bits() != 0 &&
109    //     !self.config().non_removable()
110    // {
111    //     warn!("SD status changed here! status:[{}]", reg.read_reg::<MCICardDetect>().bits());
112    //     card_detected();
113    // }
114
115    // handle error state
116    if dmac_events.contains(MCIDMACStatus::DMAC_ERR_INTS_MASK)
117        || events.contains(MCIRawInts::CMD_ERR_INTS_MASK)
118    {
119        handle_error_occur(&reg, events.bits(), dmac_events.bits());
120        return;
121    }
122
123    // handle cmd && data done
124    if events.contains(MCIRawInts::DTO_BIT) && events.contains(MCIRawInts::CMD_BIT) {
125        handle_cmd_done();
126        handle_data_done(events.bits(), dmac_events.bits());
127    } else if events.contains(MCIRawInts::CMD_BIT)
128    // todo Cannot get MCI instance here, temporarily unable to handle this case
129    // (events.contains(MCIRawInts::HTO_BIT) && self.cur_cmd_index() == MCI::SWITCH_VOLTAGE as isize)
130    {
131        handle_cmd_done();
132    // } else if events.contains(MCIRawInts::CMD_BIT) {
133    //     // handle cmd done
134    //     handle_cmd_done();
135    } else if events.contains(MCIRawInts::DTO_BIT) {
136        // handle data done
137        handle_data_done(events.bits(), dmac_events.bits());
138    }
139}

150pub fn handle_data_done(status: u32, dmac_status: u32) {
151    let check_status = status
152        & (
153            MCIRawInts::DTO_BIT      // Data transfer over
154            | MCIRawInts::RCRC_BIT    // Response CRC error
155            | MCIRawInts::DCRC_BIT    // Data CRC error
156            | MCIRawInts::RE_BIT      // Response error
157            | MCIRawInts::DRTO_BIT    // Data read timeout
158            | MCIRawInts::EBE_BIT     // End-bit error
159            | MCIRawInts::SBE_BCI_BIT // Start-bit error
160            | MCIRawInts::RTO_BIT
161            // Response timeout
162        )
163            .bits();
164    let check_dmac = dmac_status & (MCIDMACIntEn::AIS | MCIDMACIntEn::DU).bits();

182fn handle_error_occur(_reg: &MCIReg, status: u32, dmac_status: u32) {
183    if status & MCIRawInts::RE_BIT.bits() != 0 || status & MCIRawInts::RTO_BIT.bits() != 0 {
184        osa_event_set(SDMMC_OSA_EVENT_TRANSFER_CMD_FAIL);
185    }
186
187    if dmac_status & MCIDMACIntEn::DU.bits() != 0
188        || status & MCIRawInts::DCRC_BIT.bits() != 0
189        || status & MCIRawInts::RCRC_BIT.bits() != 0
190    {
191        osa_event_set(SDMMC_OSA_EVENT_TRANSFER_DATA_FAIL);
192    }
193}
