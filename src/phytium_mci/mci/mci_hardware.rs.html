36    pub(crate) fn clock_set(&self, enable: bool) {
37        let reg = self.config.reg();
38        if enable {
39            reg.set_reg(MCIClkEn::CCLK_ENABLE);
40        } else {
41            reg.clear_reg(MCIClkEn::CCLK_ENABLE);
42        }
43    }
44
45    pub(crate) fn update_exteral_clk(&self, uhs_reg: MCIClkSrc) -> MCIResult {
46        let reg = self.config.reg();
47        reg.write_reg(MCIClkSrc::empty());
48        reg.write_reg(uhs_reg);
49        reg.retry_for(
50            |reg: MCIClkSts| reg.contains(MCIClkSts::READY),
51            Some(RETRIES_TIMEOUT),
52        )?;
53        Ok(())
54    }
55
56    pub(crate) fn init_external_clk(&self) -> MCIResult {
57        let reg_val = MCIClkSrc::uhs_reg(0, 0, 0x5) | MCIClkSrc::UHS_EXT_CLK_ENA;
58        if 0x502 == reg_val.bits() {
59            info!("invalid uhs config");
60        }
61        self.update_exteral_clk(reg_val)?;
62        Ok(())
63    }

97    pub(crate) fn ctrl_reset(&self, reset_bits: MCICtrl) -> MCIResult {
98        let reg = self.config.reg();
99
100        reg.modify_reg(|reg| reset_bits | reg);
101        if let Err(e) = reg.retry_for(
102            |reg: MCICtrl| !reg.contains(reset_bits),
103            Some(RETRIES_TIMEOUT),
104        ) {
105            error!("Reset failed, bits = 0x{:x}", reset_bits);
106            return Err(e);
107        }
108
109        /* update clock after reset */
110        if let Err(e) = self.private_cmd_send(MCICmd::UPD_CLK, 0) {
111            error!("Update clock failed!");
112            return Err(e);
113        }
114
115        /* for fifo reset, need to check if fifo empty */
116        if reset_bits.contains(MCICtrl::FIFO_RESET)
117            && let Err(e) = reg.retry_for(
118                |reg: MCIStatus| reg.contains(MCIStatus::FIFO_EMPTY),
119                Some(RETRIES_TIMEOUT),
120            )
121        {
122            error!("Fifo not empty!");
123            return Err(e);
124        }
125        Ok(())
126    }

128    pub(crate) fn cardreset_config(&self) {
129        let reg = self.config.reg();
130        if self.config.non_removable() {

137    pub(crate) fn clear_interrupt_status(&self) {
138        let reg = self.config.reg();
139
140        /* Clear interrupt enable bits */
141        reg.write_reg(MCIIntMask::empty());
142
143        /* Clear current interrupt status */
144        let reg_val = reg.read_reg::<MCIRawInts>();
145        reg.write_reg(reg_val);
146
147        /* Clear DMAC interrupt enable */
148        reg.write_reg(MCIDMACIntEn::empty());
149
150        /* Clear DMAC interrupt status */
151        let reg_val = reg.read_reg::<MCIDMACStatus>();
152        reg.write_reg(reg_val);
153    }
154
155    pub(crate) fn descriptor_set(&self, descriptor: usize) {
156        let reg = self.config.reg();
157        let hi = (descriptor >> 32) as u32;
158        reg.write_reg(MCIDescListAddrH::from_bits_truncate(hi));
159        reg.write_reg(MCIDescListAddrL::from_bits_truncate(descriptor as u32));
160    }

162    pub(crate) fn idma_reset(&self) {
163        let reg = self.config.reg();
164        reg.set_reg(MCIBusMode::SWR); /* Write 1 to soft reset idma, hardware auto-clears to 0 when reset completes */
165    }
166
167    pub(crate) fn set_ddr_mode(&self, enable: bool) {
168        let reg = self.config.reg();
169        let mut uhs_val = reg.read_reg::<MCIUhsReg>();
170        let mut emmc_val = reg.read_reg::<MCIEmmcDdrReg>();
171
172        if enable {
173            uhs_val.insert(MCIUhsReg::DDR);
174            emmc_val.insert(MCIEmmcDdrReg::CYCLE);
175        } else {
176            uhs_val.remove(MCIUhsReg::DDR);
177            emmc_val.remove(MCIEmmcDdrReg::CYCLE);
178        }
179
180        reg.write_reg(uhs_val);
181        reg.write_reg(emmc_val);
182    }
