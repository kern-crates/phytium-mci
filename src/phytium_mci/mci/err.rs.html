1//! Error types for MCI operations
2//!
3//! This module defines the error types and result type used throughout
4//! the MCI driver for error handling.
5
6use super::RegError;
7
8/// Errors that can occur during MCI operations
9#[derive(Debug)]
10pub enum MCIError {
11    /// Operation timed out
12    Timeout,
13    /// Device not initialized
14    NotInit,
15    /// Buffer too short for operation
16    ShortBuf,
17    /// Operation not supported
18    NotSupport,
19    /// Device in invalid state
20    InvalidState,
21    /// Data transfer timeout
22    TransTimeout,
23    /// Command timeout
24    CmdTimeout,
25    /// No card detected
26    NoCard,
27    /// Device busy
28    Busy,
29    /// DMA buffer not properly aligned
30    DmaBufUnalign,
31    /// Invalid timing configuration
32    InvalidTiming,
33}
34
35impl RegError for MCIError {
36    fn timeout() -> Self {
37        MCIError::Timeout
38    }
39}
40
41/// Result type for MCI operations
42pub type MCIResult<T = ()> = Result<T, MCIError>;
