8/// Data Synchronization Barrier.
9///
10/// Ensures that all explicit memory accesses before the DSB instruction
11/// complete before any explicit memory accesses after the DSB instruction.
12/// This is essential for cache maintenance operations and ensuring
13/// memory ordering.
14///
15/// # Safety
16///
17/// This function uses inline assembly and should only be called when
18/// direct hardware access is required. The function combines both DSB
19/// (Data Synchronization Barrier) and ISB (Instruction Synchronization
20/// Barrier) to ensure full synchronization.
21///
22/// # Example
23///
24/// ```rust,ignore
25/// unsafe { dsb() };
26/// ```
27#[inline(always)]
28pub unsafe fn dsb() {
29    unsafe {
30        core::arch::asm!("dsb sy");
31        core::arch::asm!("isb sy");
32    }
33}
34
35/// Instruction Synchronization Barrier.
36///
37/// Ensures that all instructions in the pipeline before the ISB
38/// instruction complete before fetching any subsequent instructions.
39/// This is used to ensure instruction stream consistency after
40/// modifying system registers or changing memory mappings.
41///
42/// # Safety
43///
44/// This function uses inline assembly and should only be called when
45/// direct hardware access is required.
46///
47/// # Example
48///
49/// ```rust,ignore
50/// unsafe { isb() };
51/// ```
52#[inline(always)]
53pub unsafe fn isb() {
54    unsafe {
55        core::arch::asm!("isb", options(nostack, preserves_flags));
56    }
57}
